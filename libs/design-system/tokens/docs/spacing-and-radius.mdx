export const scale = (prefix) =>
  Object.values(tokens).filter((token) => token.id.startsWith(prefix));

export const swatch = 'var(--gdg-color-blue-500)';

export const SpacingScale = () => (
  <table style={{ width: '100%' }}>
    <thead>
      <tr>
        <th style={{ textAlign: 'left' }}>Figma</th>
        <th style={{ textAlign: 'left' }}>px</th>
        <th style={{ textAlign: 'left', width: '50%' }}>Size</th>
        <th style={{ textAlign: 'left' }}>Token</th>
      </tr>
    </thead>
    <tbody>
      {scale('spacing.').map((token) => (
        <tr key={token.id}>
          <td>{token.figma?.name}</td>
          <td>{remToPx(token.value)}</td>
          <td>
            <div
              style={{
                width: cssVar(token.id),
                height: '1rem',
                background: swatch,
              }}
            />
          </td>
          <td>
            <code>{cssVar(token.id)}</code>
          </td>
        </tr>
      ))}
    </tbody>
  </table>
);

export const RadiusScale = () => (
  <div style={{ display: 'flex', gap: '2rem', flexWrap: 'wrap' }}>
    {scale('radius.').map((token) => (
      <figure key={token.id} style={{ margin: 0, textAlign: 'center' }}>
        <div
          style={{
            width: '6rem',
            height: '6rem',
            borderRadius: cssVar(token.id),
            background: swatch,
          }}
        />
        <figcaption>
          {token.figma?.name}
          <br />
          <code>{cssVar(token.id)}</code>
        </figcaption>
      </figure>
    ))}
  </div>
);

# Spacing & Radius

Generated from the W3C design tokens in
[`@gdg-wroclaw/design-system-tokens`](https://github.com/AdrianRomanski/gdg-wroclaw/tree/main/libs/design-system/tokens/tokens)
(see ADR-0008 and ADR-0010), sourced from the Figma `spacing-*` and `radius-*` variables. Values are
authored in px and emitted in rem.

```css
.card {
  padding: var(--gdg-spacing-24);
  gap: var(--gdg-spacing-16);
  border-radius: var(--gdg-radius-12);
}
```
