import { Meta, ColorPalette, ColorItem } from '@storybook/addon-docs/blocks';
import { tokens } from '@gdg-wroclaw/shared-ui-tokens';

<Meta title="Design System/Foundations/Colors" />

export const colorTokens = Object.values(tokens).filter(
  (token) => token.type === 'color',
);

export const byFigmaGroup = (group) =>
  Object.fromEntries(
    colorTokens
      .filter((token) => token.figma?.group === group)
      .map((token) => [token.figma.name, token.value]),
  );

export const semantic = (prefix) =>
  Object.fromEntries(
    colorTokens
      .filter((token) => token.id.startsWith(prefix))
      .map((token) => [token.cssVariable, `var(${token.cssVariable})`]),
  );

# Colors

Generated from the W3C design tokens in
[`@gdg-wroclaw/shared-ui-tokens`](https://github.com/AdrianRomanski/gdg-wroclaw/tree/main/libs/shared/ui-tokens/tokens)
(see ADR-0008), sourced from the Figma **Colors** page. Prefer **semantic** tokens in components;
use palette tokens only when no semantic token fits.

```css
.card {
  background: var(--gdg-color-background-default);
  color: var(--gdg-color-content-default);
}
```

## Semantic

The product is dark by default: `background-default` is Black 02 and `content-default` is OFF White.
Brand **primary** colors are the resting state and **secondary** (halftone) colors the hover state.
